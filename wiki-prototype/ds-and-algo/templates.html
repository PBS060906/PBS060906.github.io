快速排序

void quickSort(int a[], int l, int r) {
    if (l >= r) return;
    int i = l - 1, j = r + 1, x = a[l + r >> 1];
    while (i < j) {
        while (a[++i] < x);
        while (a[--j] > x);
        if (i < j) swap(a[i], a[j]);
    }
    quickSort(a, l, j), quickSort(a, j + 1, r);
}

Dijkstra 最短路

vector<int> dijkstra(int s) {
    priority_queue<PII, vector<PII>, greater<>> pq;
    dist.assign(n, INF); dist[s] = 0;
    pq.push({0, s});
    while (!pq.empty()) {
        auto [d, u] = pq.top(); pq.pop();
        if (d > dist[u]) continue;
        for (auto [v, w] : g[u])
            if (dist[u] + w < dist[v]) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
    }
    return dist;
}

进阶模板

以下模板覆盖更复杂的场景，建议先手敲一遍，再上 OJ 用对拍数据验证。

树状数组（单点加 + 前缀和）

// 适用：单点修改、区间和查询，均为 O(log n)
int tr[N];
int lowbit(int x) { return x & -x; }
void add(int x, int v) {              // 位置 x 加上 v
    for (; x <= n; x += lowbit(x)) tr[x] += v;
}
int query(int x) {                    // 前缀和 [1, x]
    int s = 0;
    for (; x; x -= lowbit(x)) s += tr[x];
    return s;
}
// 区间和 [l, r] = query(r) - query(l - 1)

线段树（区间和）

// 适用：单点改 + 区间查；区间改需另加懒标记，此处省略
struct SegTree {
    long long sum[N << 2];
    void pushup(int u) { sum[u] = sum[u << 1] + sum[u << 1 | 1]; }
    void build(int u, int l, int r, int *a) {
        if (l == r) { sum[u] = a[l]; return; }
        int mid = l + r >> 1;
        build(u << 1, l, mid, a); build(u << 1 | 1, mid + 1, r, a);
        pushup(u);
    }
    void update(int u, int l, int r, int p, int v) {   // 点 p 加 v
        if (l == r) { sum[u] += v; return; }
        int mid = l + r >> 1;
        if (p <= mid) update(u << 1, l, mid, p, v);
        else update(u << 1 | 1, mid + 1, r, p, v);
        pushup(u);
    }
    long long query(int u, int l, int r, int ql, int qr) {  // 区间和
        if (ql <= l && r <= qr) return sum[u];
        int mid = l + r >> 1; long long s = 0;
        if (ql <= mid) s += query(u << 1, l, mid, ql, qr);
        if (qr > mid) s += query(u << 1 | 1, mid + 1, r, ql, qr);
        return s;
    }
};

拓扑排序（Kahn BFS 版）

// 适用：DAG 依赖排序；出队数不足 n 则存在环
vector<int> g[N]; int deg[N], q[N];
bool toposort(int n) {
    int hh = 0, tt = -1;
    for (int i = 1; i <= n; i++)
        if (!deg[i]) q[++tt] = i;     // 入度 0 的点先入队
    while (hh <= tt) {
        int u = q[hh++];              // 出队顺序即拓扑序
        for (int v : g[u])
            if (--deg[v] == 0) q[++tt] = v;
    }
    return tt == n - 1;               // false 表示有环
}

KMP 字符串匹配

// nxt[i]：模式串 p[1..i] 的最长相等前后缀长度；单串匹配 O(n+m)
for (int i = 2, j = 0; i <= m; i++) {  // 求 nxt 数组
    while (j && p[i] != p[j + 1]) j = nxt[j];
    if (p[i] == p[j + 1]) j++;
    nxt[i] = j;
}
for (int i = 1, j = 0; i <= n; i++) {  // 在文本串 s 中匹配 p
    while (j && s[i] != p[j + 1]) j = nxt[j];
    if (s[i] == p[j + 1]) j++;
    if (j == m) { /* s 中匹配终点为 i */ j = nxt[j]; }
}

二分图匹配（匈牙利算法）

// 适用：二分图最大匹配，复杂度 O(n·m)；match[v] 为 v 当前的配对点
int match[N]; bool vis[N];
bool dfs(int u) {
    for (int v : g[u])
        if (!vis[v]) {
            vis[v] = true;            // 本轮已访问，防止死循环
            if (!match[v] || dfs(match[v])) { match[v] = u; return true; }
        }
    return false;
}
int ans = 0;                          // 左部每个点尝试匹配一次
for (int i = 1; i <= n; i++) {
    memset(vis, false, sizeof vis);
    ans += dfs(i);
}

Warning

  • 大数据量务必开 long long 与快速 I/O（ios::sync_with_stdio(false)）；
  • 递归 DFS 注意爆栈，可改显式栈。
